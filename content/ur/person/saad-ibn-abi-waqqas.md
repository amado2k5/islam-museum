## تعارف
سعد بن ابی وقاص ابتدائی مسلمانوں میں سے تھے، بدر کے تیر انداز تھے، اور وہ سپہ سالار جنہوں نے قادسیہ میں ساسانی سلطنت کے لشکر کو شکست دی۔ کوفہ کی بنیاد بھی انہی کے ہاتھوں رکھی گئی۔

## زندگی
قریش کے نوجوان [[group/banu-zuhra|بنو زہرہ]] سے تھے، جو نبی کی والدہ کا اپنا قبیلہ تھا۔ وہ [[person/abu-bakr|ابو بکر]] کے ہاتھ پر اسلام لائے، اور اس وقت ان کی عمر بیس سال سے کم تھی، اور وہ ابتدائی دس ایمان لانے والوں میں شامل تھے {cite:ibn-sad-tabaqat-bewley}. وہ [[event/battle-of-badr|بدر]] میں شریک ہوئے، جسے روایت یہ کہتی ہے کہ انہوں نے اللہ کی راہ میں سب سے پہلا تیر چلایا، اور اس کے بعد مدینہ کی جنگوں میں بھی شریک رہے {cite:ibn-sad-tabaqat-bewley}.

سنہ 636 عیسوی میں [[person/umar-ibn-al-khattab|عمر]] کے نائب کی حیثیت سے انہوں نے [[event/battle-of-al-qadisiyya|قادسیہ]] میں عراق کے مقام پر ساسانی لشکر کا سامنا کیا اور کئی دنوں کی لڑائی کے بعد اسے شکست دی، جس کے بعد فارس کے علاقے کھل گئے {cite:tabari-donner-1993}. عمر کے حکم سے انہوں نے کوفہ کی بنیاد رکھی اور عراق کے والی بنے۔ بعد میں وہ مدینہ کے قریب عقیق میں الگ ہو گئے اور تقریباً 674 عیسوی میں وفات پائی {cite:ibn-sad-tabaqat-bewley}.

## دائرۃ المعارف اسلامیہ میں نام اور نسب
دائرۃ المعارف اسلامیہ، دوسرا ایڈیشن، میں سعد بن ابی وقاص پر ج. ر. ہاؤٹنگ کا مضمون انھیں نبی کے ممتاز صحابی اور عراق کی فتح کے دوران عرب لشکروں کا سپہ سالار بتاتا ہے۔ ان کا قبیلہ قریش کی شاخ [[group/banu-zuhra|بنو زہرہ]] تھا۔ ان کی کنیت ابو اسحاق تھی، اور انھیں سعد بن مالک کے نام سے بھی درج کیا جاتا ہے، کیونکہ ان کے والد مالک بن وہیب، جنھیں اہیب بھی کہا گیا، ابن عبد مناف بن زہرہ تھے۔ مضمون کہتا ہے کہ مالک کی کنیت ابو وقاص ہونے کی کوئی وضاحت نظر نہیں آتی۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

ایک روایت میں سعد نے نبی سے پوچھا کہ وہ کون ہیں، تو انھیں جواب ملا کہ وہ سعد بن مالک زہری ہیں۔ چونکہ نبی کی والدہ بھی بنو زہرہ سے تھیں، کہا جاتا ہے کہ نبی نے سعد کو اپنا ماموں تسلیم کیا۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

## سنی تراجم کی کتابوں کی روایات
مضمون کے مطابق سعد کو [[concept/ashara-mubashshara|جنت کی بشارت پانے والے دس صحابہ]] میں شمار کیا جاتا ہے۔ سنی تراجم کی کتابوں میں ان کے اندراجات زیادہ تر کئی موضوعات پر روایات پر مشتمل ہیں۔ ان میں ان کے ابتدائی اسلام کا ذکر ہے، کہ وہ تیسرے، ساتویں یا نویں شخص تھے، اس وقت جب نماز ابھی فرض نہیں ہوئی تھی۔ ان میں قرآن کی بعض آیات کے نزول میں ان کے کردار کا ذکر ہے، اور اس کا کہ وہ اسلام کی خاطر خون بہانے والے پہلے اور اللہ کی راہ میں تیر چلانے والے پہلے شخص تھے، اور [[event/hijra|ہجرت]] کے فوراً بعد کی رات نبی کی پہرے داری کا۔ یہ بھی مذکور ہے کہ وہ نبی کی تمام جنگوں میں شریک ہوئے۔ ان کتابوں کے مطابق نبی نے صرف ان سے، اور ایک اور روایت میں ان سے اور [[person/zubayr-ibn-al-awwam|زبیر]] سے، کہا کہ ان پر میرے ماں باپ قربان، اور نبی نے دعا کی کہ سعد کی ہر دعا قبول ہو۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

مضمون مزید کہتا ہے کہ شیعہ روایت میں صحابہ کو، سعد سمیت، عموماً زیادہ منفی نظر سے دیکھا جاتا ہے، اور زیدی اور امامی نقطہ ہائے نظر پر کولبرگ کی تحقیقات کا حوالہ دیتا ہے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

## بیماری اور وصیت کی روایات
روایات کا ایک سلسلہ نبی کی اس عیادت کا ذکر کرتا ہے جو انھوں نے مکہ میں سعد کی کی، جب وہ بیمار اور بظاہر قریب المرگ تھے، فتح کے بعد کسی وقت، اور موقع کے تعین میں اختلاف ہے۔ یہ روایات جزوی طور پر اس پر ہیں کہ سعد کو اس جگہ مرنا ناپسند تھا جہاں سے وہ ہجرت کر چکے تھے، اور جزوی طور پر نبی سے منسوب اس فیصلے پر کہ مسلمان اپنے ترکے کا کتنا حصہ وصیت کر سکتا ہے۔ مضمون تفصیل کے لیے مارسڈن سپائٹ اور ڈی. ایس. پاورز کی تحقیقات کا حوالہ دیتا ہے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

## عراق میں سپہ سالار
مضمون کے مطابق قادسیہ سے پہلے کی جنگ میں عربوں کی شکست کے بعد خلیفہ [[person/umar-ibn-al-khattab|عمر بن الخطاب]] نے مبینہ طور پر سعد کو لشکر کی کمان دے کر وسطی عراق بھیجا۔ سعد اس سے پہلے [[group/hawazin|ہوازن]] سے صدقے کا ٹیکس وصول کرنے کے ذمہ دار تھے۔ اسی لشکر نے [[event/battle-of-al-qadisiyya|قادسیہ]] میں [[polity/sasanian-empire|ساسانیوں]] کو شکست دی۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

مضمون ایک روایت درج کرتا ہے کہ سعد خود اس وقت بیمار تھے اور لڑائی میں شریک نہ ہوئے، اور بعض مآخذ ان کی غیر حاضری کے حوالے سے ان پر تنقید کرنے والے اشعار نقل کرتے ہیں۔ اس فتح سے ساسانی عراق سے نکل گئے اور سعد نے [[place/al-madain|مدائن]] پر قبضہ کیا۔ اس کی توثیق ساسانیوں کی ایک اور شکست سے ہوئی جو [[event/battle-of-jalula|جلولاء]] میں ایک دستے نے دی جسے سعد نے اپنے بھتیجے [[person/hashim-ibn-utba|ہاشم بن عتبہ بن ابی وقاص]] کی قیادت میں بھیجا تھا۔ ان واقعات کی ترتیب زمانی غیر یقینی ہے، مگر عموماً انھیں 14 سے 19ھ/635 سے 640ء کے درمیان رکھا جاتا ہے، اور مضمون ابتدائی اسلامی فتوحات پر ڈونر کی تحقیق کا حوالہ دیتا ہے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

## کوفہ کی بنیاد اور معزولی
فتح کے ساتھ [[place/kufa|کوفہ]] کی بنیاد پڑی، جو ان فوجوں کی چھاؤنی تھی جو قادسیہ میں لڑی تھیں اور بعد میں مدائن میں ٹھہری تھیں۔ مضمون کہتا ہے کہ نئے شہر کی بنیاد کی ہدایات کے بارے میں کہا جاتا ہے کہ وہ خود عمر کی طرف سے آئیں، جبکہ بستی کو منظم کرنے کی ذمہ داری سعد سے منسوب ہے، اور وہ اس کے پہلے والی بنے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

پھر عمر نے انھیں معزول کر دیا، بظاہر کوفہ والوں کی شکایات کے بعد۔ ان پر جو الزامات لگائے جانے کا ذکر ہے ان میں سب سے نمایاں یہ تھا کہ وہ نماز ٹھیک طرح نہیں پڑھاتے تھے، اور یہ بھی کہ وہ اپنے فیصلوں میں ناانصافی کرتے تھے، مال غنیمت منصفانہ تقسیم نہیں کرتے تھے، اور مہمات کو ٹھیک طرح منظم نہیں کرتے تھے۔ مضمون اشارہ کرتا ہے کہ کوفہ میں سعد کے مکان کی آرائش یا عیش کی روایات بھی متعلق ہو سکتی ہیں، جسے کہا جاتا ہے کہ عمر نے ناپسند کیا اور جلوا دیا۔ بعض روایات سے ظاہر ہوتا ہے کہ سعد کو بعد میں عمر کے دور میں اور غالباً [[person/uthman-ibn-affan|عثمان]] کے دور میں بھی کوفہ پر اختیار ملا، مگر تفصیلات غیر یقینی ہیں۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749}

## شوریٰ اور خلافت کا سوال
مضمون کہتا ہے کہ بڑے پیمانے پر روایت ہے کہ معزولی کے باوجود عمر نے سعد کو ان چھ صحابہ میں نامزد کیا جو [[event/shura-council-of-uthman|شوریٰ]] میں تھے، جسے انھوں نے 23ھ/644ء میں اپنے جانشین کے انتخاب کے لیے مقرر کیا۔ مضمون خبردار کرتا ہے کہ بلاذری کی انساب الاشراف واقدی، موسیٰ بن عقبہ اور زہری کی سند پر سعد کی رکنیت کی صراحتاً تردید کرتی ہے۔ ایک کثیر المتداول روایت میں عمر نے تقرری کے وقت کہا کہ انھوں نے سعد کو کوفہ سے کسی کمزوری یا خیانت کی وجہ سے نہیں ہٹایا، اور اگر سعد خلیفہ چنے جائیں تو یہ قبول کیا جائے، ورنہ جو بھی چنا جائے وہ سعد سے مدد لے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 749} {cite:encyclopaedia-of-islam-2 vol. 8 p. 750}

## علی اور معاویہ کے درمیان فتنہ
ابتدائی اسلام کا آخری اہم واقعہ جس سے مضمون سعد کو جوڑتا ہے وہ [[person/ali-ibn-abi-talib|علی بن ابی طالب]] اور [[person/muawiya-ibn-abi-sufyan|معاویہ بن ابی سفیان]] کی کشمکش ہے۔ کہا جاتا ہے کہ سعد غیر جانبدار رہے۔ بعض روایات میں جب ان سے کسی فریق کا ساتھ دینے کو کہا گیا تو انھوں نے جواب دیا کہ وہ تب کریں گے جب انھیں ایسی تلوار دی جائے جو مومن اور کافر میں فرق کر سکے۔ مضمون کہتا ہے کہ اس غیر جانبداری کو کبھی کبھی زاہدانہ کنارہ کشی کی ایک صورت کے طور پر پیش کیا جاتا ہے، اور کہا جاتا ہے کہ انھوں نے اپنے لیے خلافت کا کوئی دعویٰ پیش نہیں کیا حالانکہ اسلام میں ان کا مقام اس کا جواز دیتا تھا۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 750}

مآخذ اس بارے میں متضاد ہیں کہ آیا وہ تحکیم میں شریک ہوئے یا نہیں۔ بعض روایات کے مطابق انھوں نے عثمان کے قتل کے بعد علی کی بیعت نہیں کی، اور دوسری کے مطابق انھوں نے فتنے کے خاتمے کے بعد آخرکار معاویہ کی بیعت کی۔ یہاں مضمون کا متن جزوی طور پر ناخواندہ ہے، اس لیے یہ بیان نہیں کیا جاتا کہ اس سے پہلے انھوں نے کیا کیا تھا۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 750}

## آخری برس اور وفات
کہا جاتا ہے کہ سعد نے اپنی زندگی کا آخری حصہ مدینہ کے قریب عقیق میں اپنے مکان میں گزارا۔ وفات پر انھیں وہاں سے مدینہ لایا گیا اور [[place/baqi-al-gharqad|بقیع]] کے قبرستان میں دفن کیا گیا، اور والی [[person/marwan-ibn-al-hakam|مروان بن الحکم]] نے ان کی نماز جنازہ پڑھائی۔ ان کی وفات کے لیے 50ھ/670 تا 671ء اور 58ھ/677 تا 678ء کے درمیان مختلف تاریخیں دی گئی ہیں، اور ان کی عمر تقریباً ستر سے اسی سال سے زیادہ تک بتائی جاتی ہے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 750}

## مآخذ کے بارے میں مضمون کا جائزہ
مضمون کے مصنف کی رائے میں غالب امکان ہے کہ تاریخی سعد کی یادیں روایات میں بہت بڑھا چڑھا کر اور ترقی دے کر بیان ہوئی ہیں۔ ان کے خیال میں ان کے بارے میں مواد غالباً تقدیسی، جدلی، فقہی اور دیگر دلچسپیوں کی عکاسی کرتا ہے، اور دلچسپ قصوں اور قیاس آرائی کی ضرورت کی بھی۔ مضمون قرآنی تفاسیر کے تین مقامات کی طرف بھی اشارہ کرتا ہے، سورہ 6، 8 اور 31 میں، جہاں سعد کو شان نزول کا سبب بتایا گیا ہے۔ {cite:encyclopaedia-of-islam-2 vol. 8 p. 750}

## اثر
کوفہ، جو علی کا دارالحکومت بنا، ابتدائی عربی زبان اور فقہ کا مرکز بنا، اور مشرق کا اہم ٹھکانہ بن گیا، اور اس کا آغاز سعد کے لشکر کے پڑاؤ سے ہوا۔ نبی سے مروی یہ دعا کہ «اے اللہ سعد کے تیر کو نشانے پر لگا» نے انہیں روایت میں مثالی تیر انداز کے طور پر مشہور کیا۔

## کم عمر قارئین کے لیے
سعد نوجوانی میں اسلام لائے، روایت کے مطابق انہوں نے «پہلا تیر» چلایا، اور بڑے ہو کر اپنے زمانے کی دو عظیم طاقتوں میں سے ایک کو شکست دی۔

## خفیہ دعوت کے زمانے میں (الرحیق المختوم، صفحہ 76)
کتاب انہیں ان لوگوں میں شمار کرتی ہے جو [[person/abu-bakr|ابو بکر]] کی دعوت پر [[event/al-dawa-al-sirriyya|خفیہ دعوت]] کے برسوں میں اسلام لائے، اور ان آٹھ افراد میں سے ایک تھے جو لوگوں سے پہلے ایمان لائے اور ابتدائی گروہ اور اسلام کا ہراول بنے {cite:mubarakpuri-raheeq p.76} {cite:ibn-hisham-sira 1/245-262}. کتاب انہیں ان کی زہری نسبت کے ساتھ ذکر کرتی ہے {cite:mubarakpuri-raheeq p.76}.

## شیخ عمر عبدالکافی کے درس سے: الدین القیم

شیخ عمر عبدالکافی سعد بن ابی وقاص کا ذکر سلسلے کے مختلف مقامات پر کرتے ہیں۔ وہ بیان کرتے ہیں کہ سعد نے لوگوں سے اللہ کی قسم دے کر پوچھا کہ مدائن کا وہ نقب کس نے کھولا تھا، تو وہ شخص معلوم نہ ہو سکا کیونکہ نقب کھول کر وہ غائب ہو گیا {cite:assabile-omar-abdelkafy al-dinu-al-qayem/ep02 00:20:00}۔

شیخ قادسیہ میں عمر بن الخطاب کے سعد کے نام خط سے استدلال کرتے ہیں: مسلمانوں کی مدد اللہ کی اطاعت سے ہوتی ہے اور دشمن کی نافرمانی سے، اور لشکر کے گناہ اس کے دشمن سے زیادہ خطرناک ہیں {cite:assabile-omar-abdelkafy al-dinu-al-qayem/ep02 00:25:00}۔ شیخ کے نزدیک (ان کی ذاتی رائے)، یہ خط بتاتا ہے کہ دین ہر چیز کو شامل ہے اور سیاست سے جدا نہیں {cite:assabile-omar-abdelkafy al-dinu-al-qayem/ep02 00:26:00}۔

شیخ بیان کرتے ہیں کہ نبی نے سعد سے فرمایا «فداك أبي وأمي»، اور یہ جملہ سعد کے سوا صرف ایک اور صحابی کے لیے کہا گیا {cite:assabile-omar-abdelkafy al-dinu-al-qayem/ep11 00:52:00}۔ وہ بیان کرتے ہیں کہ عمر نے سعد کو ان چھ افراد میں شامل کیا جن میں سے جنت کی بشارت پانے والے دس میں سے ایک کو خلیفہ چننا تھا {cite:assabile-omar-abdelkafy al-dinu-al-qayem/ep10 00:49:30}، اور یہ کہ سعد طلحہ اور زبیر کے ساتھ خلافت سے دستبردار ہوئے {cite:assabile-omar-abdelkafy al-dinu-al-qayem/ep10 00:51:00}۔

## ابو اسحاق حوینی کی «ضفاف الکوثر» میں
[[person/abu-ishaq-al-heweny|ابو اسحاق حوینی]] بتاتے ہیں کہ علی اور معاویہ کے درمیان فتنے میں صحابہ تقسیم ہو گئے، اور سعد بن ابی وقاص ان میں سے تھے جو اس سے الگ رہے: وہ مدینہ سے نکل گئے، ذو الحلیفہ کے پاس مبارک وادی عقیق میں اپنے لیے گھر بنایا اور لوگوں سے کنارہ کش ہو گئے۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep02 00:11:00} {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep02 00:11:30} وہ اس پر سعد کی بیان کردہ ایک مثال روایت کرتے ہیں: ایک قافلہ راستے میں تھا کہ گہری دھند چھا گئی؛ کچھ دائیں گئے، کچھ بائیں، اور کچھ اپنی جگہ ٹھہرے رہے یہاں تک کہ دھند چھٹ گئی تو وہ پہلے ہی راستے پر تھے۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep02 00:12:00} حوینی تشریح کرتے ہیں کہ جو دائیں بائیں گئے انہوں نے اجتہاد کیا اور ممکن ہے غلطی کریں، اور جو ٹھہرے رہے وہ یقیناً درست تھے، کیونکہ اس مقام تک سب متفق تھے۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep02 00:13:00}

ایک اور قسط میں وہ کہتے ہیں کہ سعد ماں کے ساتھ حسنِ سلوک میں ضرب المثل تھے، اور وہ آیت جس میں والدین کے شرک پر مجبور کرنے کی کوشش کے باوجود دنیا میں ان کے ساتھ بھلائی سے رہنے کا حکم ہے، انہی کے سبب نازل ہوئی۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep09 00:09:00} جب وہ اسلام لائے تو ان کی ماں نے قسم کھائی کہ نہ کھائے گی نہ پیے گی یہاں تک کہ مر جائے، تاکہ انہیں ماں کا قاتل کہہ کر عار دلائی جائے؛ حوینی اسے بھوک ہڑتال سے تشبیہ دیتے ہیں۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep09 00:09:30} وہ سعد سے روایت کرتے ہیں کہ جب وہ اسے کھلانا چاہتے تو لکڑی سے اس کا منہ کھولتے، اور سعد نے اس سے کہا کہ اگر اس کی سو جانیں ہوں اور ایک ایک کر کے نکلیں تب بھی وہ اپنا دین نہ چھوڑیں گے۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep09 00:10:00} {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep09 00:10:30} حوینی آیت کو یوں پڑھتے ہیں کہ وہ دین پر ثابت قدمی کے ساتھ نرم گفتاری سکھاتی ہے، اور سعد کے ماں سے کہے گئے الفاظ میں ایک سختی دیکھتے ہیں جسے آیت نے سنوارا؛ یہ ان کی قراءت ہے۔ {cite:assabile-abu-ishaq-alheweny dhifaf-al-kawtar/ep09 00:11:00}

## مسعد انور کے سلسلے «المبشرون بالجنة» میں
[[work/the-ten-promised-paradise-mosaad-anwar|«المبشرون بالجنة»]] کی چوبیسویں قسط میں [[person/mosaad-anwar|مسعد انور]] کہتے ہیں کہ سعد بن ابی وقاص اس وقت اسلام لائے جب مسلمان ایک ہاتھ کی انگلیوں پر گنے جا سکتے تھے؛ وہ واحد صحابی ہیں جن پر نبی نے اپنے ماں باپ دونوں کو فدا کہا؛ وہ مستجاب الدعوات تھے؛ اور اللہ کی راہ میں سب سے پہلے خون بہانے اور سب سے پہلے تیر چلانے والے تھے {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:01:00}؛ ان کا اسلام کی راہ میں سب سے پہلے تیر چلانا اس ریکارڈ میں ابن سعد سے بھی مذکور ہے۔ {cite:ibn-sad-tabaqat-bewley} وہ ان کا تعارف ابو اسحاق سعد بن ابی وقاص مالک بن اُہیب کے طور پر کراتے ہیں اور انہیں قادسیہ کا ہیرو اور مجوس کی آگ بجھانے والا کہتے ہیں۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:01:30} وہ کہتے ہیں کہ سعد مکہ میں ایک معزز اور مالدار گھرانے میں پلے بڑھے، لہو و لعب سے دور رہے، تیر تراشنے، کمانیں درست کرنے اور شہسواری میں لگے رہے یہاں تک کہ مکہ کے سب سے بڑے تیر انداز بن گئے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:01:30}

وہ بیان کرتے ہیں کہ عمر نے سعد کو [[place/kufa|کوفہ]] کا والی بنایا، جس کے لوگوں کو وہ بہت مسائل کھڑے کرنے والا بتاتے ہیں، اور عمر [[person/muhammad-ibn-maslama|محمد بن مسلمہ]] کو بھیجتے کہ والی کو ساتھ لے کر مسجدوں میں پھریں اور اس کے بارے میں پوچھیں؛ لوگوں نے ہر مسجد میں سعد کی تعریف کی سوائے بنو عبس کی مسجد کے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:09:30} وہاں اسامہ بن قتادہ نامی ایک شخص، جس کی کنیت ابو سعدہ تھی، کھڑا ہوا اور سعد پر الزام لگایا کہ وہ فیصلے میں انصاف نہیں کرتے، برابر تقسیم نہیں کرتے اور سریّے کے ساتھ نہیں نکلتے؛ مقرر جواب دیتے ہیں کہ سعد تیسرے مسلمان اور عراق، جنوبی ترکی، ایران کے بیشتر حصے، [[place/armenia|آرمینیا]] اور [[place/azerbaijan|آذربائیجان]] کے فاتح تھے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:10:30} ان کی رائے میں عمر نے سعد کو کمزوری یا خیانت کی وجہ سے معزول نہیں کیا بلکہ انہیں اہلِ مدینہ کے لیے ترجیح دی؛ سعد مظلوم تھے، اور وہ یاد دلاتے ہیں کہ مظلوم کی دعا اور اللہ کے درمیان کوئی پردہ نہیں۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:11:30} وہ بیان کرتے ہیں کہ سعد نے اس شخص کے چار الزاموں کے بدلے اس پر چار بددعائیں کیں: اللہ اسے اندھا کرے، اس کی عمر لمبی کرے، اس کا فقر دائمی کرے اور اسے فتنوں کے سامنے کرے؛ وہ اندھا ہوا، محتاج ہو کر بھیک مانگنے لگا، اور اتنا جیا کہ اس کی بھنویں آنکھوں پر آ گریں، اور اسی برس سے اوپر کی عمر میں لونڈیوں سے چھیڑ چھاڑ کرتا اور کہتا کہ اسے سعد کی بددعا لگ گئی ہے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:12:00} وہ کہتے ہیں کہ عمر نے وفات کے وقت سعد کو شوریٰ کے چھ افراد میں رکھا اور فرمایا کہ اگر وہ خلیفہ نہ بنیں تو جو بھی والی ہو ان سے مدد لے؛ مقرر اس سے یہ سمجھتے ہیں کہ عمر ان کی طرف اشارہ کر رہے تھے۔ پھر عثمان منتخب ہوئے تو انہوں نے سعد سے مدد لی اور انہیں دوسری بار کوفہ کا والی بنایا۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:15:00}

وہ کہتے ہیں کہ سعد نبی کے ساتھ [[event/battle-of-badr|بدر]]، احد، [[event/battle-of-the-trench|خندق]]، [[event/conquest-of-mecca|فتحِ مکہ]]، [[event/battle-of-hunayn|حنین]] اور تمام غزوات میں شریک ہوئے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:16:00} وہ بیان کرتے ہیں کہ جب عمر نے سنا کہ فارسی مدینہ پر حملے کی تیاری کر رہے ہیں تو قسم کھائی کہ عجم کے بادشاہوں پر عرب کے بادشاہوں کو پھینکیں گے؛ شہسواروں کو جمع کیا اور خود لشکر کی قیادت کرنا چاہی؛ انہیں رکنے کا مشورہ دیا گیا تو انہوں نے سعد کو چنا، جو اس وقت [[group/hawazin|ہوازن]] کے صدقات پر مامور تھے، اور انہیں شیر کہا، اور عبد الرحمن بن عوف نے ان کے وصف میں اضافہ کیا؛ اور عمر نے انہیں ہر روز خط لکھنے کا حکم دیا۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:16:30} وہ کہتے ہیں کہ ان کے شمار کے مطابق [[event/battle-of-al-qadisiyya|قادسیہ]] میں مسلمان چھتیس ہزار تھے اور فارسی دو لاکھ چالیس ہزار، جن کے ساتھ سولہ ہاتھی تھے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:17:30} وہ کہتے ہیں کہ جنگ چار دن جاری رہی اور چالیس ہزار فارسی مارے گئے، جن میں ان کا سپہ سالار [[person/rustam-farrukhzad|رستم]] بھی تھا۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:18:30} وہ بیان کرتے ہیں کہ جنگ سے پہلے سعد کو عرق النساء اور بہت سے پھوڑے نکل آئے تھے، اس لیے وہ جنگی لباس نہ پہن سکے؛ انہوں نے میدان میں ایک اور شخص کو سپہ سالار بنایا اور ایک اونچے گھر کی چھت سے احکام بھیجتے رہے؛ جب لشکر میں اس پر سرگوشیاں ہوئیں تو وہ نیچے آئے اور اپنا جسم کھول کر دکھایا، تو لوگوں نے انہیں معذور سمجھا۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:18:30} وہ کہتے ہیں کہ صحابہ نے تمام ہاتھیوں کو ان کے سواروں سمیت مار ڈالا، اور وہ قادسیہ اور [[event/battle-of-yarmuk|یرموک]] کو خلفائے راشدین کے دور کی سب سے بڑی جنگیں شمار کرتے ہیں، پہلی سعد کی قیادت میں اور دوسری [[person/khalid-ibn-al-walid|خالد]] کی قیادت میں۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:19:30} اس کے بعد سعد نے [[place/al-madain|مدائن]]، پورا [[place/iraq|عراق]]، جنوبی ترکی، ایران کا بیشتر حصہ، آرمینیا اور آذربائیجان فتح کیے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:20:00}

وہ ابو حامد [[person/al-ghazali-abu-hamid|غزالی]] کی [[work/ihya-ulum-al-din|«احیاء علوم الدین»]] کی چوتھی جلد سے روایت کرتے ہیں کہ سعد نابینا ہونے کے بعد مکہ آئے تو لوگوں نے ان سے دعا کی درخواست کی؛ مکہ کے قاری [[person/abd-allah-ibn-al-saib|عبد اللہ بن سائب]] نے پوچھا کہ وہ اپنی بینائی کی واپسی کی دعا کیوں نہیں کرتے، تو سعد نے کہا کہ اللہ کا فیصلہ انہیں اپنی بینائی سے زیادہ محبوب ہے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:21:00} وہ کہتے ہیں کہ وفات کے وقت سعد ستر برس سے اوپر تھے، اور ان کے بیٹے [[person/musab-ibn-sad-ibn-abi-waqqas|مصعب]] سے روایت کرتے ہیں کہ وہ رو رہے تھے اور باپ کا سر ان کی گود میں تھا، تو سعد نے کہا کہ مت رو، اللہ انہیں عذاب نہیں دے گا اور وہ اہلِ جنت میں سے ہیں۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:22:00} انہوں نے اون کا ایک پرانا جبہ منگوایا جسے پہن کر وہ بدر میں شریک ہوئے تھے اور جسے کفن کے لیے سنبھال رکھا تھا۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:22:30} ان کی وفات عقیق میں ہوئی، پھر انہیں [[place/medina|مدینہ]] لایا گیا، جہاں اس کے امیر [[person/marwan-ibn-al-hakam|مروان بن حکم]] نے ان کی نمازِ جنازہ پڑھائی، اور نبی کی ازواج اور ہزاروں صحابہ و تابعین نے بھی نماز پڑھی، اور وہ [[place/baqi-al-gharqad|بقیع]] میں دفن ہوئے۔ {cite:assabile-mosaad-anwar the-ten-promised-paradise/ep24 00:23:00} اس قسط میں وہ عثمان کی شہادت کے بعد فتنے میں سعد کا موقف اور علی کو برا کہنے والے ایک شخص پر ان کی بددعا کا واقعہ بھی بیان کرتے ہیں، جو یہاں نہیں لکھا گیا۔
